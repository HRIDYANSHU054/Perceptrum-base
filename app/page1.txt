"use client";

import { useCallback, useMemo, useRef, useState } from "react";

import { BoxCoords, HandPosition } from "@/types/types";
import HandRecognizer from "@/components/HandRecognizer";
import MovingHands from "@/components/MovingHands";
import Drum from "@/components/Drum";
import Sidebar from "@/components/Sidebar/Sidebar";
import { DrumsProvider } from "@/contexts/DrumsContext";
import DrumsContainer from "@/components/DrumsContainer";

function Page() {
  const videoContainerRef = useRef<HTMLDivElement>(null);

  const [leftHandLeft, setLeftHandLeft] = useState(1000); // or the x coord
  const [rightHandLeft, setRightHandLeft] = useState(1000);

  const [leftHandTop, setLeftHandTop] = useState(500); // or the x coord
  const [rightHandTop, setRightHandTop] = useState(500);

  //for passing to the drums for detecting collisions
  const leftHandCoords = useRef<BoxCoords>({
    left: 0,
    right: 0,
    top: 0,
    bottom: 0,
  });
  const rightHandCoords = useRef<BoxCoords>({
    left: 0,
    right: 0,
    top: 0,
    bottom: 0,
  });

  //mempozed object to pass as props
  const leftHandPosition = useMemo(
    () => ({ left: leftHandLeft, top: leftHandTop }),
    [leftHandLeft, leftHandTop]
  );
  const rightHandPosition = useMemo(
    () => ({ left: rightHandLeft, top: rightHandTop }),
    [rightHandLeft, rightHandTop]
  );
  //%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

  //also memoize this function.
  const setHandResults = useCallback(function (
    lhPos: HandPosition,
    rhPos: HandPosition,
    isDetected: boolean
  ) {
    if (isDetected && videoContainerRef.current) {
      const containerRect = videoContainerRef.current.getBoundingClientRect();
      const containerLeft = containerRect.left;
      const containerTop = containerRect.top;
      const containerRight = containerRect.right;
      const containerBottom = containerRect.bottom;

      const updateHandPosition = (
        pos: HandPosition,
        setLeft: (value: number) => void,
        setTop: (value: number) => void
      ) => {
        if (pos.left || pos.top) {
          const newLeft = pos.left * window.innerWidth;
          const newTop = pos.top * window.innerHeight;
          const iconSize = 48;

          // Check if the new position is within bounds
          if (
            newLeft >= containerLeft &&
            newLeft <= containerRight - iconSize
          ) {
            setLeft(newLeft); // Adjust to container's coordinate system
          }
          if (newTop > containerTop && newTop < containerBottom - iconSize) {
            setTop(newTop); // Adjust to container's coordinate system
          }
        }
      };

      updateHandPosition(lhPos, setLeftHandLeft, setLeftHandTop);
      updateHandPosition(rhPos, setRightHandLeft, setRightHandTop);

      // console.log("Detection was done");
      // console.log(leftHandPosition, rightHandPosition);
    }
  },
  []);

  return (
    <DrumsProvider>
      <main className="h-full flex flex-col justify-center items-center w-screen ">
        {/* sidebare */}
        <Sidebar />

        <div
          className="relative w-2/3 aspect-video  rounded-lg overflow-hidden border-4 border-stone-400"
          ref={videoContainerRef}
        >
          <HandRecognizer
            leftHandPosition={leftHandPosition}
            rightHandPosition={rightHandPosition}
            setHandResults={setHandResults}
          />

          {/* Drums Container */}
          {/* REMEMBER this container is also positioned set */}
          {/* will appear above the video and will take as much space */}


          <div className="top-0 left-0 absolute z-5 w-full h-full bg-blue-700/10 text-white flex items-center justify-center ">
            DRUMS
            <Drum
              lhCoords={leftHandCoords.current}
              rhCoords={rightHandCoords.current}
              constrainToParent={videoContainerRef}
            />
            <Drum
              initXPos={0}
              initYPos={70}
              lhCoords={leftHandCoords.current}
              rhCoords={rightHandCoords.current}
              constrainToParent={videoContainerRef}
              flip={true}
            />
          </div>
          
        </div>

        {/* hands */}
        <MovingHands
          lhPos={leftHandPosition}
          rhPos={rightHandPosition}
          rhCoordsRef={rightHandCoords}
          lhCoordsRef={leftHandCoords}
        />
      </main>
    </DrumsProvider>
  );
}

export default Page;
